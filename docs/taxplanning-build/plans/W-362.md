**Row 4 — the routes, a sign-in, a refresh and a logout (§1.i, §1.j, items 6 and 8).**
```bash
B=http://localhost:8080
curl -s -o /dev/null -w 'rights %{http_code}\n' "$B/api/v1/public/privacy/rights"
curl -s "$B/api/v1/public/privacy/rights" | jq -r '"notice version " + .version'
curl -s -o /dev/null -w 'old rights path %{http_code}\n' "$B/privacy/rights"
LOGIN=$(curl -s -X POST "$B/api/v1/public/auth/login" -H 'Content-Type: application/json' -d "$(jq -n --arg e "$OWNER_EMAIL" --arg p "$APP_PASSWORD" '{email:$e,password:$p}')")
export TOKEN=$(jq -r .access_token <<<"$LOGIN") REFRESH=$(jq -r .refresh_token <<<"$LOGIN")
printf 'expires_in %s; refresh token %s chars\n' "$(jq -r .expires_in <<<"$LOGIN")" "${#REFRESH}"
curl -s -o /dev/null -w 'my stocks %{http_code}\n' -H "Authorization: Bearer $TOKEN" "$B/api/v1/me/portfolio/stocks"
curl -s -o /dev/null -w 'basic auth %{http_code}\n' -u "$OWNER_EMAIL:$APP_PASSWORD" "$B/api/v1/me/portfolio/stocks"
curl -s -H "Authorization: Bearer $TOKEN" "$B/api/v1/admin/docs/openapi" | jq -r '"spec paths " + (.paths | length | tostring)'
NEXT=$(curl -s -X POST "$B/api/v1/public/auth/token/refresh" -H 'Content-Type: application/json' -d "$(jq -n --arg r "$REFRESH" '{refresh_token:$r}')")
NEXT_REFRESH=$(jq -r .refresh_token <<<"$NEXT")
printf 'refreshed expires_in %s; rotated %s\n' "$(jq -r .expires_in <<<"$NEXT")" "$([ -n "$NEXT_REFRESH" ] && [ "$NEXT_REFRESH" != "$REFRESH" ] && echo yes || echo no)"
curl -s -o /dev/null -w 'logout %{http_code}\n' -X POST "$B/api/v1/public/auth/logout" -H 'Content-Type: application/json' -d "$(jq -n --arg r "$NEXT_REFRESH" '{refresh_token:$r}')"
```
